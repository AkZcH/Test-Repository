$$
	\text{Productivity} = \frac{\text{Output produced}}{\text{input used}}
$$

$$
	\text{Partial Productivity} = \frac{\text{Output}}{\text{input of one class}}
$$

$$
\text{Efficiency} = \frac{\text{Actual output produced}}{\text{Ideal Output Expected}}
$$

$$
	\text{Net Output} = \text{Total Output} - \text{Input Used}
$$

$$
\text{Service Productivity} = \frac{\text{Revenue generated by the service}}{\text{Cost of producing the service}}
$$
